func check(e error) { if e!=nil { panic(e) } }
func emit(m map[string]any) { check(json.NewEncoder(os.Stdout).Encode(m)) }
func identity() map[string]any {
 tid:=syscall.Gettid();s,e:=os.ReadFile(fmt.Sprintf("/proc/%d/status",tid));check(e);st,e:=os.ReadFile(fmt.Sprintf("/proc/%d/stat",tid));check(e)
 return map[string]any{"pid":os.Getpid(),"tid":tid,"status":string(s),"stat":string(st)}
}
func checkpoint(tag string, more map[string]any) {

 syscall.RawSyscall(155,0,0,0)
}
