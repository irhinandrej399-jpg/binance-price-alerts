# OPS-062: provenance scope без transport checkpoint

Диагностический probe для сессионной проверки, не реализация runtime.
Сохранить Go-блок во временный
`internal/matcher/pricework/ops062_probe_test.go`, затем выполнить:

```sh
GOCACHE=/tmp/binance-go-build python3 scripts/check_postgres.py
```

Runner работает только с disposable `binance-go-check` и удаляет его volume.
После проверки удалить временный Go-файл. Не использовать рабочую БД.
Новый scope и тот же source без cursor должны давать `pgx.ErrNoRows`.
Другой source должен отличаться ошибкой mismatch, legacy — unbound.
Текущий ReadCheckpoint не различает эти случаи; Pending пуст во всех четырёх.
Падение последних двух subtests — ожидаемое воспроизведение пробела интерфейса.
После OPS-063 адаптировать вызов к новому read-only API, сохранив запрет
синтетического Kafka offset. Сам ReadCheckpoint может сохранить контракт.

```go
package pricework_test

import (
 "context"
 "errors"
 "testing"

 w "github.com/irhinandrej399-jpg/binance-price-alerts/internal/matcher/pricework"
 "github.com/irhinandrej399-jpg/binance-price-alerts/internal/matcher/sourcebinding"
 "github.com/irhinandrej399-jpg/binance-price-alerts/internal/platform/kafkasource"
 "github.com/irhinandrej399-jpg/binance-price-alerts/internal/testutil/postgrestest"
 "github.com/jackc/pgx/v5"
)

// Temporary readiness probe; retained verbatim in the OPS-062 handoff.
func TestOPS062BootstrapProvenanceProbe(t *testing.T) {
 p := postgrestest.OpenWithDSN(t, "MATCHER_DATABASE_URL")
 ctx := context.Background()
 for _, mode := range []string{"empty", "same", "different", "legacy"} {
  t.Run(mode, func(t *testing.T) {
   src := priceSource()
   scope := t.Name()
   if mode == "legacy" {
    in := input("latest", 0)
    in.Position = nil
    old := w.New(p, scope)
    if err := old.Accept(ctx, in); err != nil { t.Fatal(err) }
    if err := old.Complete(ctx, in.Key, nil, nil, nil); err != nil { t.Fatal(err) }
   }
   g := sourceGrant(t, p, scope)
   if mode == "same" || mode == "different" {
    old := w.NewSourceBound(p, g, src)
    in := sourceInput(g, src, "latest", 0)
    in.Position = nil
    if err := old.Accept(ctx, in); err != nil { t.Fatal(err) }
    if err := old.Complete(ctx, in.Key, nil, nil, nil); err != nil { t.Fatal(err) }
   }
   want := pgx.ErrNoRows
   if mode == "different" { src.TopicID[0] = 2; want = kafkasource.ErrMismatch }
   if mode == "legacy" { want = sourcebinding.ErrUnbound }
   store := w.NewSourceBound(p, g, src)
   keys, err := store.Pending(ctx)
   if err != nil || len(keys) != 0 { t.Fatal(keys, err) }
   _, err = store.ReadCheckpoint(ctx, "prices", 2)
   if !errors.Is(err, want) { t.Fatalf("bootstrap provenance: got %v, want %v; Pending empty", err, want) }
  })
 }
}
```
